Кастомні помилки дозволяють точно ідентифікувати тип проблеми в програмі. Стандарт: розширювати Error через class, задавати name, додавати специфічні поля. Це дає instanceof, стек і семантику.

Базовий кастомний Error

JavaScript
class ValidationError extends Error {
  constructor(message, field) {
    super(message);
    this.name  = 'ValidationError'; // обов'язково перевизначити
    this.field = field;
  }
}

class HttpError extends Error {
  constructor(statusCode, message) {
    super(message);
    this.name       = 'HttpError';
    this.statusCode = statusCode;
  }
}

// Використання
function readUser(data) {
  if (!data.name) throw new ValidationError('Name required', 'name');
  if (!data.email) throw new ValidationError('Email required', 'email');
  return data;
}

try {
  readUser({ name: 'Alice' });
} catch (e) {
  if (e instanceof ValidationError) {
    console.log(`Поле: ${e.field}, Помилка: ${e.message}`);
  } else {
    throw e; // решта — прокидаємо вгору
  }
}

Ієрархія помилок

JavaScript
// Базовий клас для всіх помилок застосунку
class AppError extends Error {
  constructor(message, code) {
    super(message);
    this.name = 'AppError';
    this.code = code;
  }
}

class DatabaseError extends AppError {
  constructor(message, query) {
    super(message, 'DB_ERROR');
    this.name  = 'DatabaseError';
    this.query = query;
  }
}

class NotFoundError extends AppError {
  constructor(resource, id) {
    super(`${resource} with id=${id} not found`, 'NOT_FOUND');
    this.name     = 'NotFoundError';
    this.resource = resource;
    this.id       = id;
  }
}

// Обробка по ієрархії
try {
  throw new NotFoundError('User', 42);
} catch (e) {
  if (e instanceof NotFoundError) {
    console.log(e.resource, e.id); // 'User' 42
  } else if (e instanceof AppError) {
    console.log('App error:', e.code);
  } else {
    throw e;
  }
}

Wrapping errors

JavaScript
// Обгортання низькорівневих помилок
class ReadError extends Error {
  constructor(message, cause) {
    super(message);
    this.name  = 'ReadError';
    this.cause = cause; // оригінальна помилка (ES2022: стандартна опція)
  }
}

function readUser(json) {
  let data;
  try {
    data = JSON.parse(json);
  } catch (e) {
    throw new ReadError('Parse failed', e);
  }
  if (!data.name) throw new ValidationError('Name required', 'name');
  return data;
}

try {
  readUser('{ bad json }');
} catch (e) {
  if (e instanceof ReadError) {
    console.log(e.message);      // 'Parse failed'
    console.log(e.cause.message); // 'Unexpected token b...'
  }
}

// ES2022: вбудований cause
try {
  JSON.parse('bad');
} catch (e) {
  throw new Error('High-level message', { cause: e });
}

Патерн wrapping: низькорівнева помилка (JSON, DB, HTTP) загортається у бізнес-помилку з контекстом. Споживач бачить зрозуміле повідомлення, а cause зберігає технічні деталі для логування.

Підсумок

  • Завжди задавай this.name у конструкторі — інакше буде 'Error'
  • Будуй ієрархію: AppError → DomainError → SpecificError
  • Додавай специфічні поля (field, statusCode, query) для контексту
  • Wrapping: зберігай cause при перекиданні помилок між шарами
  • Прокидай (throw e) невідомі помилки — не ковтай мовчки
