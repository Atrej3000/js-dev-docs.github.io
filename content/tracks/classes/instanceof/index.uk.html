Оператор instanceof перевіряє, чи належить об'єкт до певного класу через ланцюжок прототипів. Поведінку можна налаштувати через Symbol.hasInstance.

Базова поведінка

JavaScript
class Animal {}
class Dog extends Animal {}

const dog = new Dog();

console.log(dog instanceof Dog);    // true
console.log(dog instanceof Animal); // true — через ланцюжок прототипів
console.log(dog instanceof Object); // true — всі об'єкти

// Як це працює всередині:
// dog.__proto__ === Dog.prototype          ✓ → true
// dog.__proto__.__proto__ === Animal.prototype ✓ → true

Перевірка через прототипи

JavaScript
// instanceof перевіряє весь ланцюжок до null
function checkChain(obj, Cls) {
  let proto = Object.getPrototypeOf(obj);
  while (proto !== null) {
    if (proto === Cls.prototype) return true;
    proto = Object.getPrototypeOf(proto);
  }
  return false;
}

// Еквівалентно instanceof
console.log(checkChain(dog, Dog));    // true
console.log(checkChain(dog, Animal)); // true

// Пастка: зміна prototype після створення
function Foo() {}
const foo = new Foo();
Foo.prototype = {};        // перезаписали прототип

console.log(foo instanceof Foo); // false! — foo досі посилається на старий prototype

Symbol.hasInstance

JavaScript
// Кастомна логіка instanceof
class EvenNumber {
  static [Symbol.hasInstance](num) {
    return typeof num === 'number' && num % 2 === 0;
  }
}

console.log(2  instanceof EvenNumber); // true
console.log(3  instanceof EvenNumber); // false
console.log(10 instanceof EvenNumber); // true

instanceof vs typeof vs constructor

JavaScript
const arr = [1, 2, 3];
const obj = {};

// typeof
typeof arr;      // 'object' — не розрізняє масив і об'єкт
typeof null;     // 'object' — відома помилка

// instanceof
arr instanceof Array;   // true
arr instanceof Object;  // true
null instanceof Object; // false — не кидає помилку

// Array.isArray — найнадійніший спосіб для масивів
Array.isArray(arr);     // true
Array.isArray(obj);     // false

// constructor (крихке — легко замінити)
arr.constructor === Array;   // true (але можна перезаписати)

// Object.prototype.toString — універсальний тег
Object.prototype.toString.call(arr);  // '[object Array]'
Object.prototype.toString.call(null); // '[object Null]'
Object.prototype.toString.call(/re/); // '[object RegExp]'

Для перевірки примітивів (string, number, boolean) instanceof завжди повертає false. Використовуй typeof для примітивів і instanceof для класів та об'єктів.

Підсумок

  • instanceof перевіряє ланцюжок прототипів аж до null
  • Повертає false для примітивів
  • Symbol.hasInstance дозволяє задати кастомну логіку перевірки
  • Для масивів — Array.isArray(); для тегів — Object.prototype.toString.call()
