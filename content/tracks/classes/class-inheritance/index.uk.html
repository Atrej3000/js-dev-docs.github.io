extends створює ланцюг прotototype між класами. super — посилання на батьківський клас: super() у конструкторі викликає батьківський конструктор, super.method() — батьківський метод.

extends і super()

JavaScript
class Animal {
  constructor(name) {
    this.name = name;
  }
  speak() { return `${this.name} makes a sound`; }
  toString() { return `[Animal: ${this.name}]`; }
}

class Dog extends Animal {
  constructor(name, breed) {
    super(name);   // ОБОВ'ЯЗКОВО перед this у конструкторі нащадка
    this.breed = breed;
  }

  // Перевизначення методу
  speak() { return `${this.name} barks`; }

  // Виклик батьківського через super
  describe() { return `${super.toString()}, breed: ${this.breed}`; }
}

const rex = new Dog('Rex', 'Labrador');
console.log(rex.speak());    // "Rex barks" — перевизначений
console.log(rex.describe()); // "[Animal: Rex], breed: Labrador"
console.log(rex instanceof Dog);    // true
console.log(rex instanceof Animal); // true — ланцюг прotototype

Правила super

JavaScript
class Base {
  constructor() { this.type = 'base'; }
  method() { return 'base method'; }
}

class Derived extends Base {
  constructor() {
    // super() ОБОВ'ЯЗКОВИЙ до будь-якого this у нащадку
    super();
    this.extra = 'derived'; // тільки після super()
  }

  method() {
    const base = super.method(); // батьківський метод
    return `${base} + derived`;
  }
}

// Якщо constructor не визначено — автоматично
class AutoConstructor extends Base {
  // constructor(...args) { super(...args); } — вставляється неявно
}
const a = new AutoConstructor();
console.log(a.type); // "base"

Приватні поля і наслідування

JavaScript
class Account {
  #balance = 0;

  deposit(amount) { this.#balance += amount; }
  get balance()   { return this.#balance; }
}

class SavingsAccount extends Account {
  #interestRate;

  constructor(rate) {
    super();
    this.#interestRate = rate;
  }

  addInterest() {
    // this.#balance — SyntaxError: приватне поле недоступне в нащадку!
    // Але метод deposit — успадкований і доступний
    this.deposit(this.balance * this.#interestRate);
  }
}

const acc = new SavingsAccount(0.05);
acc.deposit(1000);
acc.addInterest();
console.log(acc.balance); // 1050

Підсумок

  • extends встановлює прotototype-ланцюг між класами.
  • super() у конструкторі нащадка — обов'язковий до this.
  • super.method() — виклик батьківського методу.
  • Приватні поля # не успадковуються — доступні лише в класі де визначені.
