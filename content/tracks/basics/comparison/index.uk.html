JavaScript має два типи перевірки рівності: строге (===) і нестроге (==). Різниця між ними є джерелом цілого класу важко відтворюваних помилок. Строге порівняння — стандарт у сучасному коді.

Реляційні оператори

Повертають boolean. Для чисел — числове порівняння. Для рядків — лексикографічне.

JavaScript
console.log(5 > 3);    // true
console.log(5 < 3);    // false
console.log(5 >= 5);   // true
console.log(5 <= 4);   // false

// Рядки: порівняння по кодах Unicode символ за символом
console.log('B' > 'A');      // true  (66 > 65)
console.log('abc' > 'ab');   // true  (довший виграє за рівних)
console.log('apple' > 'Banana'); // true  — мала 'a' > велика 'B'
// Увага: 'a'.charCodeAt(0) = 97, 'B'.charCodeAt(0) = 66

// Порівняння різних типів → числова конвертація
console.log('5' > 3);    // true  ('5' → 5)
console.log(null > 0);   // false
console.log(null == 0);  // false
console.log(null >= 0);  // true  — несподівано! (null → 0 у >=)

Строге порівняння ===

Оператор === перевіряє і значення, і тип без будь-яких перетворень. Якщо типи різні — завжди false.

JavaScript
// Однакові типи і значення
console.log(1 === 1);         // true
console.log('abc' === 'abc'); // true
console.log(true === true);   // true
console.log(null === null);   // true

// Різні типи — завжди false
console.log(1 === '1');       // false
console.log(0 === false);     // false
console.log(null === undefined); // false

// Об'єкти: порівнюються за посиланням
const a = { x: 1 };
const b = { x: 1 };
const c = a;
console.log(a === b); // false — різні об'єкти в пам'яті
console.log(a === c); // true  — одне і те ж посилання

// Спеціальні значення
console.log(NaN === NaN);   // false — NaN не дорівнює нічому
console.log(+0 === -0);     // true  — === не розрізняє -0 і 0

Нестроге порівняння ==

Оператор == виконує Abstract Equality Comparison зі специфікації ECMAScript: при різних типах застосовує перетворення перед порівнянням. Результати часто неінтуїтивні.

JavaScript
// Число і рядок: рядок конвертується в число
console.log(1 == '1');      // true  ('1' → 1)
console.log(0 == '');       // true  ('' → 0)
console.log(0 == '0');      // true  ('0' → 0)
console.log('' == '0');     // false (обидва рядки, різні)

// Boolean конвертується в число
console.log(false == 0);    // true  (false → 0)
console.log(true == 1);     // true  (true → 1)
console.log(true == '1');   // true  (обидва → 1)
console.log(false == '');   // true  (false→0, ''→0)

// null і undefined: рівні тільки між собою
console.log(null == undefined);  // true  — спеціальне правило
console.log(null == 0);          // false — null не конвертується
console.log(null == false);      // false
console.log(undefined == false); // false

// Об'єкт і примітив: ToPrimitive
console.log([] == false);    // true  ([] → '' → 0, false → 0)
console.log([] == 0);        // true  ([] → '' → 0)
console.log([1] == 1);       // true  ([1] → '1' → 1)
console.log('' == false);    // true  ('' → 0, false → 0)
== — джерело помилок

Нестроге порівняння має 11 правил перетворення. Навіть досвідчені розробники іноді помиляються з ними. Використовуйте === за замовчуванням. Єдиний прийнятний виняток: перевірка на null або undefined одночасно: value == null.

Особливості null та undefined

JavaScript
// == між null і undefined — true (спеціальне правило)
console.log(null == undefined);   // true
console.log(null === undefined);  // false

// null у реляційних операторах: конвертується в 0
console.log(null > 0);    // false  (0 > 0 = false)
console.log(null < 0);    // false  (0 < 0 = false)
console.log(null >= 0);   // true   (0 >= 0 = true)
console.log(null == 0);   // false  — УВАГА: == НЕ конвертує null!
// null == 0 та null >= 0 дають суперечливі результати

// undefined у будь-яких операціях → NaN
console.log(undefined > 0);   // false
console.log(undefined < 0);   // false
console.log(undefined == 0);  // false

// Практична перевірка: == null покриває обидва випадки
function process(value) {
  if (value == null) {   // true для null і undefined
    return 'empty';
  }
  return value;
}

Object.is — строге без виключень

Object.is(a, b) аналогічний ===, але коректно обробляє два крайніх випадки:

JavaScript
// === та Object.is
console.log(NaN === NaN);        // false
console.log(Object.is(NaN, NaN)); // true ← правильно

console.log(+0 === -0);          // true
console.log(Object.is(+0, -0));  // false ← розрізняє

// В решті випадків — ідентичний ===
console.log(Object.is(1, 1));    // true
console.log(Object.is(1, '1')); // false

Підсумок

  • Реляційні оператори > < >= <=: числове порівняння, для рядків — лексикографічне по Unicode.
  • === — строге: не виконує перетворень, різні типи завжди false.
  • == — нестроге: виконує Abstract Equality Comparison. Результати часто неінтуїтивні. Уникайте.
  • null == undefined — true (спеціальне правило). Це єдиний доцільний кейс для ==.
  • null у реляційних операторах конвертується в 0, але null == 0 — false.
  • Object.is(a, b): строге порівняння без двох виключень === — правильно обробляє NaN і -0.
  • Об'єкти порівнюються за посиланням: два різних об'єкти з однаковим вмістом — не рівні.
