Цикли дозволяють повторювати блок коду певну кількість разів або поки виконується умова. JavaScript має кілька видів циклів.

Цикл while

Цикл while виконує тіло, поки умова є true.

let i = 0;
while (i < 3) {
  console.log(i); // 0, 1, 2
  i++;
}

// Якщо умова хибна з початку — тіло не виконується
let n = 10;
while (n < 5) {
  console.log(n); // не виконається жодного разу
}

Цикл do...while

Цикл do...while виконує тіло хоча б один раз, а потім перевіряє умову.

let i = 0;
do {
  console.log(i); // 0, 1, 2
  i++;
} while (i < 3);

// Навіть якщо умова хибна з початку — тіло виконується один раз
let n = 10;
do {
  console.log(n); // 10 — виконається один раз
} while (n < 5);

Цикл for

Найпопулярніший цикл. Має три частини: початок, умову і крок.

// for (початок; умова; крок)
for (let i = 0; i < 3; i++) {
  console.log(i); // 0, 1, 2
}

// Частини можна пропускати
let i = 0;
for (; i < 3; i++) {
  console.log(i);
}

// Нескінченний цикл
for (;;) {
  // зупинити через break
}

break і continue

break — повністю зупиняє цикл. continue — пропускає поточну ітерацію.

// break: зупинитись при i === 3
for (let i = 0; i < 10; i++) {
  if (i === 3) break;
  console.log(i); // 0, 1, 2
}

// continue: пропустити парні
for (let i = 0; i < 6; i++) {
  if (i % 2 === 0) continue;
  console.log(i); // 1, 3, 5
}

Мітки для вкладених циклів

Мітка дозволяє break або continue вийти із зовнішнього циклу.

outer: for (let i = 0; i < 3; i++) {
  for (let j = 0; j < 3; j++) {
    if (j === 1) continue outer; // перейти до наступної ітерації зовнішнього циклу
    console.log(i, j);
  }
}
// 0 0
// 1 0
// 2 0

for...of та for...in

// for...of — ітерація значень (масиви, рядки, Map, Set)
const arr = ['a', 'b', 'c'];
for (const val of arr) {
  console.log(val); // a, b, c
}

// for...in — ітерація ключів об'єкта (не рекомендується для масивів)
const obj = { x: 1, y: 2, z: 3 };
for (const key in obj) {
  console.log(key, obj[key]); // x 1, y 2, z 3
}

Підсумок

  • while — повторюй поки умова true
  • do...while — як while, але завжди виконується хоч раз
  • for (init; cond; step) — класичний цикл з лічильником
  • break — вийти з циклу, continue — пропустити ітерацію
  • for...of — для ітерабельних значень, for...in — для ключів об'єкта
