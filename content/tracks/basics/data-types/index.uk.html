ECMAScript визначає 8 типів даних: 7 примітивних і один складний. JavaScript — мова з динамічною типізацією: тип визначається значенням під час виконання і може змінюватись. Знання всіх типів та їх граничних значень є базою для розуміння перетворень, порівнянь і поведінки операторів.

Number

Єдиний числовий тип для цілих і дробових значень. Реалізований як 64-бітний формат IEEE 754 double-precision. Безпечний діапазон цілих: від -(2⁵³ - 1) до 2⁵³ - 1.

JavaScript
const integer   = 42;
const float     = 3.14159;
const negative  = -7;
const separator = 1_000_000;    // числовий роздільник (ES2021)
const hex       = 0xFF;         // 255 у шістнадцятковій
const octal     = 0o377;        // 255 у вісімковій
const binary    = 0b11111111;   // 255 у двійковій
const exp       = 1.5e3;        // 1500

// Спеціальні значення
const inf     = Infinity;
const negInf  = -Infinity;
const nan     = NaN;            // Not a Number — результат невалідних операцій

console.log(1 / 0);             // Infinity
console.log(-1 / 0);            // -Infinity
console.log(0 / 0);             // NaN
console.log('text' - 1);        // NaN

// NaN не дорівнює нічому, включно з собою
console.log(NaN === NaN);       // false
console.log(Number.isNaN(NaN)); // true — правильна перевірка

Точність IEEE 754

JavaScript
console.log(0.1 + 0.2);          // 0.30000000000000004 — IEEE 754
console.log(0.1 + 0.2 === 0.3);  // false

// Коректне порівняння дробів
const EPSILON = Number.EPSILON;
console.log(Math.abs(0.1 + 0.2 - 0.3) < EPSILON); // true

// Безпечна межа цілих
console.log(Number.MAX_SAFE_INTEGER); // 9007199254740991
console.log(9007199254740991 + 1);    // 9007199254740992 — ОК
console.log(9007199254740991 + 2);    // 9007199254740992 — помилка точності!

String

Незмінна послідовність кодових одиниць UTF-16. Нема окремого типу «символ» — одиночний символ є рядком довжиною 1.

JavaScript
const single   = 'одинарні лапки';
const double   = "подвійні лапки";
const template = `шаблонний літерал: ${1 + 2}`;   // "шаблонний літерал: 3"

// Екранування
const withQuote = 'it\'s fine';
const newline   = "рядок 1\nрядок 2";
const tab       = "col1\tcol2";
const unicode   = "\u0041";     // "A"

// Рядки незмінні: методи повертають новий рядок
const str = "  Hello  ";
console.log(str.trim());           // "Hello"
console.log(str.toLowerCase());    // "  hello  "
console.log(str.trim().length);    // 5

// Шаблонний літерал — багаторядковий без \n
const html = `
  <div>
    <p>Вміст</p>
  </div>
`;

Boolean

Рівно два значення: true і false. Є результатом порівнянь та логічних операцій.

JavaScript
const isActive  = true;
const hasErrors = false;

// Порівняння повертають boolean
console.log(5 > 3);       // true
console.log('a' === 'b'); // false

// Falsy значення: false, 0, -0, 0n, "", null, undefined, NaN
// Truthy — все решта, включно з [] і {}
console.log(Boolean(0));         // false
console.log(Boolean(''));        // false
console.log(Boolean(null));      // false
console.log(Boolean([]));        // true  — порожній масив truthy!
console.log(Boolean({}));        // true  — порожній об'єкт truthy!

null

null — примітив, що явно позначає «відсутність значення». Встановлюється програмістом навмисно.

JavaScript
let currentUser = null; // явно: «зараз нікого не авторизовано»

// typeof null — відомий баг специфікації (з 1995 р.)
console.log(typeof null);   // "object" — НЕ "null"

// Перевірка на null: завжди суворе порівняння
if (currentUser === null) {
  redirectToLogin();
}

// Нестрога перевірка — обидва null і undefined
if (currentUser == null) {
  // спрацює для null і undefined
}

undefined

undefined означає, що змінна оголошена, але значення не присвоєно. Також є результатом читання неіснуючої властивості або відсутнього аргументу функції.

JavaScript
let status;
console.log(status);            // undefined — не ініціалізовано

const obj = { name: 'admin' };
console.log(obj.age);           // undefined — властивість не існує

function greet(name) {
  console.log(name);            // undefined якщо не передано аргумент
}
greet();

// Явне присвоєння undefined — антипатерн
// Використовуйте null для явного "немає значення"
let data = undefined; // погано
let data2 = null;     // краще

Symbol

Унікальний незмінний примітив. Кожен Symbol() — гарантовано новий унікальний ідентифікатор. Призначений для створення унікальних ключів властивостей без ризику колізій.

JavaScript
const id1 = Symbol('id');
const id2 = Symbol('id');

console.log(id1 === id2);  // false — завжди унікальні

// Символи як ключі об'єкта — не конфліктують зі строковими ключами
const KEY = Symbol('key');
const obj = {
  key: 'string key',
  [KEY]: 'symbol key',
};

console.log(obj.key);   // "string key"
console.log(obj[KEY]);  // "symbol key"

// Символи не потрапляють у for...in та Object.keys()
console.log(Object.keys(obj)); // ["key"] — без Symbol

// Symbol.for — глобальний реєстр (той самий символ за описом)
const s1 = Symbol.for('shared');
const s2 = Symbol.for('shared');
console.log(s1 === s2); // true

BigInt

Цілі числа довільної точності. Суфікс n або функція BigInt(). Необхідний для значень за межами Number.MAX_SAFE_INTEGER.

JavaScript
const huge   = 9_007_199_254_740_993n;  // більше MAX_SAFE_INTEGER
const fromFn = BigInt(9007199254740991);

console.log(typeof huge); // "bigint"
console.log(huge + 1n);   // 9007199254740994n

// BigInt і Number не можна змішувати напряму
// console.log(1n + 1); // TypeError
console.log(1n + BigInt(1)); // 2n
console.log(Number(1n));     // 1 — конвертація в Number

// BigInt не має Infinity і NaN
// console.log(1n / 0n); // RangeError

Object

Єдиний складний тип. Все, що не є примітивом — об'єкт: літеральні об'єкти, масиви, функції, дати, регулярні вирази, Map, Set тощо. Об'єкти передаються за посиланням, примітиви — за значенням.

JavaScript
// Літеральний об'єкт
const config = { host: 'localhost', port: 3000 };

// Масив — спеціалізований об'єкт
const items = ['a', 'b', 'c'];
console.log(Array.isArray(items)); // true — правильна перевірка
console.log(typeof items);         // "object" — не "array"

// Функція — callable object
function fn() {}
console.log(typeof fn); // "function" — спеціальний підтип Object

// Передача за посиланням
const a = { value: 1 };
const b = a;
b.value = 2;
console.log(a.value); // 2 — a і b посилаються на один об'єкт

// Передача примітиву — за значенням (копія)
let x = 1;
let y = x;
y = 2;
console.log(x); // 1 — x не змінився

Edge cases

JavaScript
// NaN не дорівнює нічому
console.log(NaN === NaN); // false
console.log(NaN !== NaN); // true — тест на NaN: Number.isNaN()

// Перевірка масиву: typeof не допоможе
console.log(typeof []);         // "object"
console.log(Array.isArray([])); // true — правильно

// Перевірка null: не typeof
console.log(typeof null);       // "object"
console.log(null === null);     // true — правильно

// -0 є у JavaScript
console.log(-0 === 0);          // true — оператор === не розрізняє
console.log(Object.is(-0, 0)); // false — Object.is розрізняє

// Порожній рядок і 0 — falsy, але не рівні
console.log('' == 0);  // true  — нестрога рівність з перетворенням
console.log('' === 0); // false — суворий тип

Підсумок

  • 7 примітивів: Number, String, Boolean, null, undefined, Symbol, BigInt. Один складний тип: Object.
  • Number — IEEE 754 double. Є Infinity, -Infinity, NaN. Дробові числа мають похибку: 0.1 + 0.2 !== 0.3.
  • NaN !== NaN. Перевіряйте через Number.isNaN().
  • null — явна відсутність (встановлює розробник). undefined — неініціалізоване або відсутнє.
  • typeof null === "object" — баг. Перевіряйте через === null.
  • Масив — Object. Перевіряйте через Array.isArray().
  • Примітиви передаються за значенням, Object — за посиланням.
  • Object.is(a, b) — строге порівняння без виключень: розрізняє -0 і 0, NaN дорівнює собі.
