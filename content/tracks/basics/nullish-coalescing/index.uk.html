Оператор ?? (nullish coalescing, ES2020) повертає правий операнд лише якщо лівий є null або undefined. Це точніший аналог || для ситуацій, де 0, false і "" є валідними значеннями.

Синтаксис і механіка

JavaScript
// a ?? b
// Повертає a якщо a !== null && a !== undefined
// Інакше повертає b

console.log(null ?? 'default');      // "default"
console.log(undefined ?? 'default'); // "default"
console.log(0 ?? 'default');         // 0     — 0 не є null/undefined
console.log('' ?? 'default');        // ""    — порожній рядок не замінюється
console.log(false ?? 'default');     // false
console.log(NaN ?? 'default');       // NaN

JavaScript
// || НЕПРАВИЛЬНО: замінює валідні 0 і false
function setVolume(level) {
  const volume = level || 50; // якщо level = 0 → volume = 50 (помилка!)
  return volume;
}
console.log(setVolume(0));   // 50 — неочікувано!

// ?? ПРАВИЛЬНО: замінює тільки null/undefined
function setVolumeFixed(level) {
  const volume = level ?? 50;
  return volume;
}
console.log(setVolumeFixed(0));         // 0 — правильно
console.log(setVolumeFixed(null));      // 50
console.log(setVolumeFixed(undefined)); // 50

// Практичний приклад: налаштування з дефолтами
function createServer(options = {}) {
  const port    = options.port    ?? 3000;
  const host    = options.host    ?? 'localhost';
  const timeout = options.timeout ?? 5000;
  const debug   = options.debug   ?? false;
  return { port, host, timeout, debug };
}

console.log(createServer({ port: 0, debug: false }));
// { port: 0, host: 'localhost', timeout: 5000, debug: false }
// port: 0 і debug: false збережені — ?? не замінив їх

Комбінація з ?. (optional chaining)

JavaScript
const user = null;

// ?. повертає undefined якщо ланцюг обривається
// ?? підставляє значення за замовчуванням
const city = user?.address?.city ?? 'Невідомо';
console.log(city); // "Невідомо"

// Без ?. і ??
const cityOld = user && user.address && user.address.city || 'Невідомо';
// Проблема: якщо city = "" → поверне 'Невідомо' (помилка)

??= — nullish присвоєння (ES2021)

JavaScript
// a ??= b  →  a = a ?? b
let config = null;
config ??= { timeout: 5000 };
console.log(config); // { timeout: 5000 }

config ??= { timeout: 9999 }; // не змінює — config вже не null
console.log(config); // { timeout: 5000 }

// Ін'єкція дефолтів в об'єкт
const settings = { theme: 'dark', fontSize: null };
settings.fontSize ??= 16;
settings.theme    ??= 'light'; // не змінює — 'dark' не null
console.log(settings); // { theme: 'dark', fontSize: 16 }

Обмеження: не можна змішувати з || і &&

JavaScript
// SyntaxError: Cannot mix ?? with && or ||
// null || undefined ?? 'default'; // помилка

// Правильно: дужки для явного порядку
(null || undefined) ?? 'default'; // "default"
null || (undefined ?? 'default'); // "default"

Підсумок

  • ?? повертає правий операнд тільки якщо лівий є null або undefined.
  • Відмінність від ||: 0, "", false, NaN НЕ замінюються.
  • Використовуйте ?? для значень за замовчуванням де 0, false або "" є валідними.
  • ??= — присвоєння тільки якщо поточне значення null/undefined.
  • Не можна змішувати ?? з || або && без дужок.
