Оператор ... виконує дві ролі: у параметрах функції — rest (збирає залишкові аргументи у масив), у виразах — spread (розгортає iterable у окремі значення).

Rest-параметри

JavaScript
// ...rest збирає решту аргументів у масив
function sum(...numbers) {
  return numbers.reduce((acc, n) => acc + n, 0);
}
console.log(sum(1, 2, 3, 4, 5)); // 15

// Rest після звичайних параметрів
function log(level, ...messages) {
  console.log(`[${level}]`, messages.join(' '));
}
log('INFO', 'Server', 'started', 'on', 'port', '3000');
// [INFO] Server started on port 3000

// Rest ЗАВЖДИ останній параметр
// function bad(...a, b) {} // SyntaxError

// Відмінність від arguments
function old() {
  // arguments — array-like, немає методів масиву, є у всіх функціях
  console.log(Array.from(arguments));
}
function modern(...args) {
  // args — справжній масив, є всі методи
  console.log(args.map(x => x * 2));
}
modern(1, 2, 3); // [2, 4, 6]

Spread у масивах

JavaScript
const arr1 = [1, 2, 3];
const arr2 = [4, 5, 6];

// Конкатенація без concat
const combined = [...arr1, ...arr2];         // [1, 2, 3, 4, 5, 6]
const withExtra = [0, ...arr1, 99, ...arr2]; // [0, 1, 2, 3, 99, 4, 5, 6]

// Копія (shallow)
const copy = [...arr1];
copy.push(99);
console.log(arr1); // [1, 2, 3] — оригінал незмінний

// Spread як аргументи функції
const nums = [3, 1, 4, 1, 5, 9];
console.log(Math.max(...nums));  // 9
console.log(Math.min(...nums));  // 1

// Spread рядка у масив символів
const chars = [..."hello"]; // ['h','e','l','l','o']

// Spread Set → унікальний масив
const unique = [...new Set([1, 2, 2, 3])]; // [1, 2, 3]

Spread в об'єктах

JavaScript
const defaults = { theme: 'dark', lang: 'uk', timeout: 5000 };
const override = { lang: 'en', debug: true };

// Злиття (пізніші перезаписують ранні)
const config = { ...defaults, ...override };
// { theme: 'dark', lang: 'en', timeout: 5000, debug: true }

// Поверхнева копія
const copy = { ...config };

// Додавання/перезаписування властивості
const updated = { ...user, role: 'admin', updatedAt: Date.now() };

// Видалення властивості через деструктуризацію + spread
const { password, ...safeUser } = user; // safeUser без password

Підсумок

  • Rest (...args) у параметрах — збирає залишкові аргументи у справжній масив. Завжди останній.
  • Rest краще ніж arguments: справжній масив, є всі методи, є у стрілкових.
  • Spread (...iter) у масивах — розгортає iterable. Копія, конкатенація, аргументи функцій.
  • Spread в об'єктах — поверхневе злиття. Порядок важливий: пізніший ключ перезаписує ранній.
