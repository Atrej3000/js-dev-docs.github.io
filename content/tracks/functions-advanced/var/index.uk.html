var — старий спосіб оголошення змінних до ES6. Відрізняється від let/const у трьох ключових аспектах: функціональна (не блокова) область видимості, hoisting і дозволяє повторне оголошення. У сучасному коді використовуйте const/let.

Функціональна область видимості

JavaScript
// var не обмежується блоком — видима у всій функції
function example() {
  if (true) {
    var x = 10;  // не блокова!
    let y = 20;  // блокова
  }
  console.log(x); // 10 — var видима за межами if-блоку
  // console.log(y); // ReferenceError — let — ні
}

// var у циклі — небезпечно
for (var i = 0; i < 3; i++) { /* ... */ }
console.log(i); // 3 — витекла за межі циклу!

for (let j = 0; j < 3; j++) { /* ... */ }
// console.log(j); // ReferenceError

Hoisting з var

JavaScript
// var піднімається у верх функції, ініціалізується undefined
console.log(x); // undefined (не ReferenceError)
var x = 5;
console.log(x); // 5

// Що відбувається насправді:
// var x;          // ← підняте оголошення
// console.log(x); // undefined
// x = 5;
// console.log(x); // 5

// let/const: TDZ — використання до рядка оголошення → ReferenceError
// console.log(z); // ReferenceError: Cannot access before initialization
// let z = 5;

Повторне оголошення

JavaScript
// var дозволяє повторне оголошення (мовчки ігнорує)
var user = 'Alice';
var user = 'Bob'; // не помилка
console.log(user); // "Bob"

// let/const — SyntaxError
// let name = 'Alice';
// let name = 'Bob'; // SyntaxError: Identifier already declared

// var у глобальному scope стає властивістю window
var globalVar = 'test';
console.log(window.globalVar); // "test" (у браузері)

Коли зустрічати var

У легасі-коді var поширений. При рефакторингу:

  • Замінюйте var на const якщо змінна не переприсвоюється.
  • На let якщо потрібне переприсвоєння.
  • Перевіряйте що заміна не змінює область видимості (особливо у циклах і блоках).

Підсумок

  • var: функціональна область видимості, hoisting з undefined, повторне оголошення.
  • let/const: блокова область видимості, TDZ, заборона повторного оголошення.
  • У сучасному коді: const за замовчуванням, let коли потрібне переприсвоєння, var — ніколи.
