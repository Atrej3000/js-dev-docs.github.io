Callback — функція, передана як аргумент, яку буде викликано після завершення операції. Це перший і найстаріший підхід до асинхронності в JS. Розуміння callback-hell — основа для розуміння того, чому з'явились Promise.

Синхронні та асинхронні callbacks

JavaScript
// Синхронний callback
[1, 2, 3].forEach(n => console.log(n)); // викликається одразу

// Асинхронний callback
setTimeout(() => console.log('після затримки'), 1000);

// Класичний Node.js-стиль: error-first callback
const fs = require('fs'); // Node.js
fs.readFile('data.txt', 'utf8', (err, data) => {
  if (err) {
    console.error('Помилка:', err);
    return; // обов'язково повертаємось
  }
  console.log('Вміст:', data);
});

Callback hell

JavaScript
// Проблема: послідовні асинхронні операції → вкладеність
loadScript('script1.js', (err, script1) => {
  if (err) handleError(err);
  else loadScript('script2.js', (err, script2) => {
    if (err) handleError(err);
    else loadScript('script3.js', (err, script3) => {
      if (err) handleError(err);
      else {
        // Нарешті можна щось зробити...
        // Але вже на 5-му рівні вкладеності
      }
    });
  });
});

// Ця "піраміда приреченості" (pyramid of doom) — причина появи Promise

Error-first convention

JavaScript
// Стандарт Node.js: перший аргумент — помилка або null
function doSomething(callback) {
  try {
    const result = riskyOperation();
    callback(null, result);  // null = немає помилки
  } catch (err) {
    callback(err);           // помилка — першою
  }
}

doSomething((err, result) => {
  if (err) {
    // обробка помилки
    return;
  }
  // робота з result
});

Уникнення callback hell через іменовані функції

JavaScript
// Краще: розбити на іменовані функції
function step1(err, data) {
  if (err) return handleError(err);
  loadScript('script2.js', step2);
}
function step2(err, data) {
  if (err) return handleError(err);
  loadScript('script3.js', step3);
}
function step3(err, data) {
  if (err) return handleError(err);
  // готово
}

loadScript('script1.js', step1);
// Лінійно, але важко відстежувати потік даних

Callbacks досі широко використовуються в Node.js API, подіях DOM (addEventListener), таймерах. Але для ланцюжків операцій завжди обирай Promise або async/await.

Підсумок

  • Callback — функція, передана як аргумент для виклику після операції
  • Error-first convention: callback(err, result)
  • Callback hell виникає при послідовних асинхронних залежностях
  • Рішення: іменовані функції → Promise → async/await
