WeakMap і WeakSet — колекції зі слабкими посиланнями. Якщо об'єкт-ключ більше ніде не доступний, збирач сміття може його видалити, і запис з колекції автоматично зникне. Це критично для кешів і метаданих, що не повинні перешкоджати GC.

WeakMap

JavaScript
const cache = new WeakMap();

function processUser(user) {
  if (cache.has(user)) return cache.get(user);
  const result = { processed: true, name: user.name.toUpperCase() };
  cache.set(user, result);
  return result;
}

let user = { name: 'Alice' };
console.log(processUser(user)); // { processed: true, name: 'ALICE' }

user = null; // об'єкт більше не потрібен
// cache автоматично очищається — немає витоку пам'яті

API WeakMap

JavaScript
const wm = new WeakMap();
const key1 = {};
const key2 = {};

wm.set(key1, 'data1');
wm.set(key2, 42);

console.log(wm.has(key1));   // true
console.log(wm.get(key1));   // 'data1'
wm.delete(key1);
console.log(wm.has(key1));   // false

// WeakMap НЕ ітерабельний — немає forEach, keys(), values(), size
// Ключі — тільки об'єкти (не примітиви)

WeakMap не має методів ітерації навмисно: якщо б можна було перебрати ключі, це б перешкоджало GC — рушій не міг би знати, чи спостерігаємо ми за ключами.

WeakSet

JavaScript
const visited = new WeakSet();

function markVisited(page) {
  visited.add(page);
}

function isVisited(page) {
  return visited.has(page);
}

let page = { url: '/home' };
markVisited(page);
console.log(isVisited(page)); // true

page = null; // сторінка видалена — запис у WeakSet зникне

// API: add(obj), has(obj), delete(obj)
// Немає: size, forEach, ітерації

Сценарії застосування

JavaScript
// 1. Приватні дані об'єкта (до появи #fields)
const _private = new WeakMap();
class User {
  constructor(name) {
    _private.set(this, { password: 'secret' });
    this.name = name;
  }
  checkPassword(pwd) {
    return _private.get(this).password === pwd;
  }
}

// 2. Відстеження без утримання об'єкта
const activeListeners = new WeakSet();
function addListener(el, fn) {
  if (!activeListeners.has(el)) {
    el.addEventListener('click', fn);
    activeListeners.add(el);
  }
}

// 3. Кешування результатів (memoization)
const memoCache = new WeakMap();
function expensiveCalc(obj) {
  if (memoCache.has(obj)) return memoCache.get(obj);
  const result = /* ... */ obj.value * 2;
  memoCache.set(obj, result);
  return result;
}

Підсумок

  • WeakMap: ключ-об'єкт, слабке посилання, немає ітерації — ідеальний для кешів і приватних даних
  • WeakSet: набір об'єктів зі слабкими посиланнями — для відстеження без утримання
  • Обидві структури не заважають GC — головна перевага над Map/Set
