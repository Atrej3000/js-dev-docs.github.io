ES-модулі мають два типи експорту: named (іменований) і default (за замовчуванням). Кожен файл може мати будь-яку кількість named exports і лише один default export.

Named exports

JavaScript
// math.js — named exports
export const PI = 3.14159;
export function add(a, b) { return a + b; }
export class Vector2D {
  constructor(x, y) { this.x = x; this.y = y; }
}

// Або в кінці файлу
const E = 2.71828;
function multiply(a, b) { return a * b; }
export { E, multiply };
export { multiply as mul }; // перейменування при експорті

// Імпорт named
import { PI, add, Vector2D } from './math.js';
import { mul }               from './math.js'; // перейменований
import { add as sum }        from './math.js'; // перейменування при імпорті
import * as math             from './math.js'; // весь модуль як namespace

console.log(math.PI);   // 3.14159
console.log(math.add(1, 2)); // 3

Default export

JavaScript
// user-service.js
export default class UserService {
  async getAll() { /* ... */ }
  async getById(id) { /* ... */ }
}

// Або вираз
export default function formatDate(date) {
  return date.toISOString().slice(0, 10);
}

// Імпорт default: ім'я довільне
import UserService from './user-service.js';
import formatDate  from './format-date.js';
import MyService   from './user-service.js'; // інша назва — теж OK

// Комбінований імпорт
import defaultExport, { namedA, namedB } from './module.js';

Ре-експорт

JavaScript
// index.js — агрегуючий файл (barrel)
export { UserService } from './user-service.js';
export { ProductService } from './product-service.js';
export { OrderService } from './order-service.js';
export { formatDate, formatCurrency } from './utils.js';

// Тепер можна імпортувати з одного місця
import { UserService, formatDate } from './index.js';

Live bindings

JavaScript
// counter.js
export let count = 0;
export function increment() { count++; }

// main.js
import { count, increment } from './counter.js';
console.log(count); // 0
increment();
console.log(count); // 1 — live binding, не копія!

Підсумок

  • Named export: export const x / export { x, y }. Іменовані, кілька на файл.
  • Default export: export default fn. Один на файл. Ім'я при імпорті — довільне.
  • Ре-експорт: export { x } from './module.js' — barrel файли для зручного API.
  • Live bindings: імпортовані змінні відображають актуальне значення, не копію.
