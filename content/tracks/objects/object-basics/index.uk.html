Об'єкт — колекція пар ключ–значення. Ключі (властивості) є рядками або символами, значення — будь-який тип. Об'єкти передаються за посиланням і є основою більшості структур даних у JavaScript.

Літеральний синтаксис

JavaScript
// Порожній об'єкт
const empty = {};

// Об'єкт з властивостями
const user = {
  name: 'Олена',
  age:  28,
  isActive: true,
};

// Будь-який тип значення
const config = {
  host:    'localhost',
  port:    3000,
  tags:    ['api', 'v2'],
  timeout: null,
  connect: function() { return 'connected'; },
};

Доступ до властивостей

JavaScript
const product = { name: 'Laptop', price: 1200, 'delivery-days': 3 };

// Крапкова нотація — для ідентифікаторів
console.log(product.name);   // "Laptop"
console.log(product.price);  // 1200

// Квадратні дужки — для всіх ключів, включно з динамічними та зі спецсимволами
console.log(product['name']);            // "Laptop"
console.log(product['delivery-days']);   // 3 — через дефіс, лише bracket

// Динамічний ключ
const key = 'price';
console.log(product[key]); // 1200

// Неіснуюча властивість → undefined (не помилка)
console.log(product.weight); // undefined

// Присвоєння: нові та існуючі
product.inStock = true;    // додати нову
product.price   = 999;     // змінити існуючу

JavaScript
const name  = 'Андрій';

// Довгий синтаксис
const userLong = { name: name, email: email, role: role };

const user = { name, email, role };
// Еквівалентно { name: 'Андрій', email: 'dev@example.com', role: 'admin' }

// Скорочений метод
const api = {

  getUser: function(id) { return fetch(`${this.baseUrl}/users/${id}`); },

  // Скорочений (рекомендований)
  getUser(id) { return fetch(`${this.baseUrl}/users/${id}`); },

Обчислювані ключі

JavaScript
const field  = 'firstName';
const prefix = 'get';

// Обчислюваний ключ при оголошенні
const obj = {
  [field]: 'John',               // ключ: "firstName"
  [`${prefix}Age`]: () => 30,    // ключ: "getAge"
};

console.log(obj.firstName);  // "John"
console.log(obj.getAge());   // 30

// Динамічне формування об'єкта
function createAction(type, payload) {
  return {
    type,
    payload,
    [`${type}_timestamp`]: Date.now(),
  };
}
createAction('LOGIN', { userId: 1 });
// { type:'LOGIN', payload:{userId:1}, LOGIN_timestamp:... }

Видалення та перевірка існування

JavaScript
const settings = { theme: 'dark', lang: 'uk', beta: false };

// Видалення властивості
delete settings.beta;
console.log(settings); // { theme: 'dark', lang: 'uk' }

// Перевірка існування властивості
// АНТИПАТЕРН: порівняння з undefined — не розрізняє відсутню і undefined
console.log(settings.theme !== undefined); // true, але...
const obj2 = { key: undefined };
console.log(obj2.key !== undefined);       // false — хоча ключ існує!

// ПРАВИЛЬНО: оператор in — перевіряє існування ключа
console.log('theme' in settings); // true
console.log('beta' in settings);  // false — видалена
console.log('key' in obj2);       // true — ключ існує зі значенням undefined

// hasOwn (ES2022) — лише власні властивості (не прототип)
console.log(Object.hasOwn(settings, 'theme')); // true
console.log(Object.hasOwn(settings, 'toString')); // false — з прототипу

Обхід властивостей

JavaScript
const user = { name: 'Олена', age: 28, role: 'admin' };

// for...in: всі enumerable ключі (включно з прототипом)
for (const key in user) {
  if (Object.hasOwn(user, key)) {  // фільтруємо успадковані
    console.log(key, user[key]);
  }
}

// Object.keys() — масив власних ключів
console.log(Object.keys(user));   // ['name', 'age', 'role']

// Object.values() — масив значень
console.log(Object.values(user)); // ['Олена', 28, 'admin']

// Object.entries() — масив [ключ, значення]
for (const [key, value] of Object.entries(user)) {
  console.log(`${key}: ${value}`);
}
// name: Олена
// age: 28
// role: admin

// Перетворення об'єкт → масив → назад
const doubled = Object.fromEntries(
  Object.entries(user).map(([k, v]) => [k, typeof v === 'number' ? v * 2 : v])
);
// { name: 'Олена', age: 56, role: 'admin' }

Порядок ключів

Специфікація ECMAScript гарантує наступний порядок ітерації для Object.keys / for...in:

  1. Цілочисельні ключі (рядки що відповідають невід'ємним цілим) — у зростаючому порядку
  2. Рядкові ключі — у порядку додавання
  3. Символьні ключі — у порядку додавання
JavaScript
const mixed = {
  name:  'test',
  '2':   'two',
  '1':   'one',
  role:  'admin',
  '10':  'ten',
};

console.log(Object.keys(mixed));
// ['1', '2', '10', 'name', 'role']
// Цілочисельні ключі йдуть першими і відсортовані!
Не покладайтесь на порядок ключів об'єкта

Якщо порядок важливий — використовуйте Map. Map гарантує порядок вставки для всіх ключів без винятків.

Spread і злиття об'єктів

JavaScript
const defaults = { theme: 'light', lang: 'en', timeout: 5000 };
const override = { lang: 'uk', timeout: 3000 };

// Злиття (пізніші ключі перезаписують ранні)
const config = { ...defaults, ...override };
// { theme: 'light', lang: 'uk', timeout: 3000 }

// Копія об'єкта (shallow)
const copy = { ...config };

// Поверхнева копія — вкладені об'єкти залишаються посиланням
const original = { a: 1, nested: { b: 2 } };
const shallow  = { ...original };
shallow.nested.b = 99;
console.log(original.nested.b); // 99 — мутує оригінал!

// Object.assign — аналог spread (мутує перший аргумент)
const target = Object.assign({}, defaults, override);

// Глибока копія через structuredClone (ES2022)
const deep = structuredClone(original);
deep.nested.b = 42;
console.log(original.nested.b); // 2 — оригінал незмінний

Edge cases

JavaScript
// Зарезервовані слова — валідні ключі (у лапках не обов'язково)
const obj = { for: 1, let: 2, return: 3 };
console.log(obj.for); // 1

// Числові ключі — зберігаються як рядки
const o = { 1: 'one' };
console.log(o[1]);    // "one" — доступ через число конвертується
console.log(o['1']);  // "one"
console.log(typeof Object.keys(o)[0]); // "string"

// Ключ __proto__ — не рекомендується
const evil = { __proto__: null }; // об'єкт без прототипу
// Краще: Object.create(null) для словників без прототипу

Підсумок

  • Об'єкт — колекція ключ–значення. Ключі — рядки або Symbol.
  • Крапкова нотація для ідентифікаторів, дужкова — для динамічних ключів і спецсимволів.
  • Скорочений синтаксис { name } = { name: name }.
  • Обчислювані ключі: { [expr]: value }.
  • in перевіряє існування ключа (включно з undefined-значенням). Object.hasOwn — лише власні.
  • Порядок: цілочисельні ключі спочатку (відсортовані), потім рядкові за порядком додавання.
  • Spread { ...a, ...b } — поверхневе злиття. structuredClone — глибока копія.
