Оператор new дозволяє створювати однотипні об'єкти за шаблоном. Будь-яка звичайна функція може бути конструктором — за конвенцією їх імена починаються з великої літери.

Функція-конструктор

JavaScript
// Конвенція: PascalCase для конструкторів
function User(name, role) {
  // new автоматично:
  // 1. Створює порожній об'єкт і присвоює його this
  // 2. Встановлює prototype
  // 3. Виконує тіло функції
  // 4. Повертає this (якщо немає явного return об'єкта)

  this.name = name;
  this.role = role;
  this.isActive = true;

  this.greet = function() {
    return `Привіт, я ${this.name}`;
  };
}

const alice = new User('Alice', 'admin');
const bob   = new User('Bob',   'user');

console.log(alice.name);    // "Alice"
console.log(alice.role);    // "admin"
console.log(bob.greet());   // "Привіт, я Bob"

// alice і bob — різні об'єкти
console.log(alice === bob); // false

Механіка оператора new

JavaScript
// Що відбувається при new User('Alice', 'admin'):

// Псевдокод внутрішньої механіки:
// const obj = Object.create(User.prototype);
// User.call(obj, 'Alice', 'admin');
// return obj;

// Перевірка через instanceof
console.log(alice instanceof User); // true

// Якщо конструктор повертає об'єкт — він і буде результатом new
function SpecialUser(name) {
  this.name = name;
  return { custom: true }; // повертає інший об'єкт!
}
const s = new SpecialUser('Test');
console.log(s.name);   // undefined — повернуто { custom: true }
console.log(s.custom); // true

// Якщо return повертає примітив — ігнорується, повертається this
function NormalUser(name) {
  this.name = name;
  return 42; // ігнорується
}
const n = new NormalUser('Bob');
console.log(n.name); // "Bob"

new.target

Мета-властивість new.target містить функцію, що викликана через new, або undefined при звичайному виклику.

JavaScript
function User(name) {
  if (!new.target) {
    // Викликали без new — примусово застосовуємо
    return new User(name);
  }
  this.name = name;
}

const a = new User('Alice'); // новий об'єкт
const b = User('Bob');       // теж новий об'єкт (через new.target)
console.log(b.name);         // "Bob"

Конструктор vs class

Клас (ES6) — синтаксичний цукор поверх функції-конструктора. Внутрішньо вони схожі, але class обов'язково викликати через new (без варіантів), і методи класу не перелічуються (enumerable: false).

JavaScript
// Функція-конструктор
function UserFn(name) {
  this.name = name;
}
UserFn.prototype.greet = function() { return `Hi, ${this.name}`; };

// Еквівалентний клас
class UserClass {
  constructor(name) { this.name = name; }
  greet() { return `Hi, ${this.name}`; }
}

// Обидва дають однаковий результат
const u1 = new UserFn('Alice');
const u2 = new UserClass('Alice');
console.log(u1.greet()); // "Hi, Alice"
console.log(u2.greet()); // "Hi, Alice"

// UserFn можна викликати без new (небезпечно)
// UserClass без new → TypeError

Підсумок

  • new Fn(): створює порожній об'єкт, виконує тіло з this = obj, повертає this.
  • Якщо конструктор повертає об'єкт — він стає результатом. Примітив — ігнорується.
  • new.target — захист від виклику без new.
  • class — синтаксичний цукор; обов'язково з new, методи не enumerable.
