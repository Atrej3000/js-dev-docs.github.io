Async/await

СереднійПроміси, async/await~ 16 хв

async/await — синтаксичний цукор поверх Promise. Дозволяє писати асинхронний код у синхронному стилі. async функція завжди повертає Promise. await зупиняє виконання всередині функції до виконання Promise.

Основи

JavaScript
// async функція завжди повертає Promise
async function greet() {
  return 'hello'; // еквівалент: return Promise.resolve('hello')
}
greet().then(console.log); // "hello"

// await: зупиняє виконання всередині async функції
async function fetchUser(id) {
  const res  = await fetch(`/api/users/${id}`); // чекаємо fetch
  const user = await res.json();                // чекаємо res.json()
  return user;
}

// Порівняння з Promise ланцюжком
// .then версія:
fetchUserThen(id) {
  return fetch(`/api/users/${id}`)
    .then(res => res.json());
}

Обробка помилок

JavaScript
// try/catch для async/await
async function loadConfig() {
  try {
    const res = await fetch('/config.json');
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    return await res.json();
  } catch (e) {
    console.error('Config load failed:', e.message);
    return {}; // значення за замовчуванням
  }
}

// Або .catch на async функції
loadConfig().catch(handleError);

// Зручний helper для уникнення try/catch
async function safeAwait(promise) {
  try {
    return [null, await promise];
  } catch (error) {
    return [error, null];
  }
}

const [err, user] = await safeAwait(fetchUser(1));
if (err) { handleError(err); return; }
console.log(user.name);

Паралельне виконання

JavaScript
// АНТИПАТЕРН: послідовне await (повільно)
async function slowLoad() {
  const user    = await fetchUser(1);    // чекаємо
  const orders  = await fetchOrders(1); // потім чекаємо
  return { user, orders };
  // Загальний час = час1 + час2
}

// ПРАВИЛЬНО: паралельне через Promise.all
async function fastLoad() {
  const [user, orders] = await Promise.all([
    fetchUser(1),
    fetchOrders(1),
  ]);
  return { user, orders };
  // Загальний час = max(час1, час2)
}

// Паралельний запуск, послідовна обробка
async function processItems(items) {
  const promises = items.map(item => processItem(item)); // всі запускаємо
  const results  = await Promise.all(promises);          // чекаємо всіх
  return results;
}

Top-level await

JavaScript
// В ES-модулях await можна використовувати на верхньому рівні
// (поза async функцією)

// main.js (type="module")
const config = await fetch('/config.json').then(r => r.json());
console.log(config.version);

// Зручно для ініціалізації модуля
const db = await connectDatabase(config.dbUrl);

Підсумок

  • async функція завжди повертає Promise. await зупиняє виконання і розгортає Promise.
  • Обробка помилок: try/catch всередині async функції або .catch зовні.
  • Послідовні await — повільно. Паралельне: await Promise.all([...]).
  • Top-level await — в ES-модулях без async обгортки.
