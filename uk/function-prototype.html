F.prototype

СереднійПрототипи, наслідування~ 10 хв

Кожна функція-конструктор має властивість prototype — об'єкт, який стає [[Prototype]] для всіх об'єктів, створених через new F(). Це механізм що лежить в основі успадкування через конструктори.

Властивість prototype функції

JavaScript
function Animal(name) {
  this.name = name;
}

// Animal.prototype — об'єкт що стане прототипом екземплярів
Animal.prototype.speak = function() {
  return `${this.name} speaks`;
};

const dog = new Animal('Rex');
const cat = new Animal('Luna');

console.log(dog.speak()); // "Rex speaks"
console.log(cat.speak()); // "Luna speaks"

// Методи СПІЛЬНІ для всіх екземплярів (один об'єкт у пам'яті)
console.log(dog.speak === cat.speak); // true — один і той самий метод

// [[Prototype]] екземпляра = Animal.prototype
console.log(Object.getPrototypeOf(dog) === Animal.prototype); // true
console.log(dog instanceof Animal); // true

Властивість constructor

JavaScript
// За замовчуванням prototype = { constructor: F }
function Rabbit() {}
console.log(Rabbit.prototype.constructor === Rabbit); // true

const r = new Rabbit();
console.log(r.constructor === Rabbit); // true — через прotototype

// Якщо перезаписати prototype — constructor втрачається!
Rabbit.prototype = {
  jump() { return 'jump'; },
};
const r2 = new Rabbit();
console.log(r2.constructor === Rabbit); // false — перезаписали!
console.log(r2.constructor);            // Object

// ПРАВИЛЬНО: зберегти constructor при розширенні
Rabbit.prototype = {
  constructor: Rabbit,   // явно відновити
  jump() { return 'jump'; },
};

F.prototype vs class

JavaScript
// Конструктор + prototype (ES5)
function Person(name) { this.name = name; }
Person.prototype.greet = function() { return `Hi, ${this.name}`; };

// Еквівалентний клас (ES6)
class PersonClass {
  constructor(name) { this.name = name; }
  greet() { return `Hi, ${this.name}`; }
}

// Внутрішньо вони дають однаковий результат
// Але class забезпечує: обов'язковий new, методи non-enumerable, super

Підсумок

  • new F() встановлює [[Prototype]] = F.prototype.
  • Методи на F.prototype спільні для всіх екземплярів (економія пам'яті).
  • При перезаписі prototype об'єктом — відновлюйте constructor вручну.
  • class робить те саме автоматично і безпечніше.
