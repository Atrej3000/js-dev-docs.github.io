Функції

ПочатковийОснови JavaScript~ 14 хв

Функція — іменований, перевикористовуваний блок коду. В JavaScript функції є first-class objects: їх можна передавати як аргументи, повертати з інших функцій і присвоювати змінним.

Function declaration

JavaScript
// Синтаксис: function ім'я(параметри) { тіло }
function greet(name) {
  return `Привіт, ${name}!`;
}

console.log(greet('Олена')); // "Привіт, Олена!"

// Function declaration піднімається (hoisting) — можна викликати до оголошення
console.log(square(4)); // 16
function square(n) {
  return n * n;
}

Параметри і аргументи

Параметри — імена у визначенні функції. Аргументи — значення, передані при виклику.

JavaScript
function add(a, b) {  // a, b — параметри
  return a + b;
}
add(2, 3);            // 2, 3 — аргументи

// Зайві аргументи ігноруються
console.log(add(1, 2, 3, 4)); // 3 — третій і четвертий проігноровані

// Відсутні аргументи → undefined
function show(x, y) {
  console.log(x, y);
}
show(1); // 1 undefined

Значення параметрів за замовчуванням (ES6)

JavaScript
// Значення за замовчуванням застосовується якщо аргумент undefined
function createUser(name, role = 'user', active = true) {
  return { name, role, active };
}

console.log(createUser('Admin', 'admin')); // { name:'Admin', role:'admin', active:true }
console.log(createUser('Guest'));          // { name:'Guest', role:'user', active:true }

// Значення за замовчуванням — будь-який вираз, обчислюється при виклику
function log(message, timestamp = Date.now()) {
  console.log(timestamp, message);
}

// Явна передача undefined → спрацьовує дефолт
createUser('Bob', undefined, false);
// { name:'Bob', role:'user', active:false }
// null НЕ спрацьовує як дефолт!
createUser('Bob', null, false);
// { name:'Bob', role:null, active:false }

Оператор return

JavaScript
// return повертає значення і зупиняє виконання
function divide(a, b) {
  if (b === 0) return null; // ранній вихід
  return a / b;
}

// Функція без явного return повертає undefined
function logMessage(msg) {
  console.log(msg);
  // немає return → повертає undefined
}
const result = logMessage('test'); // undefined

// ASI пастка: return і значення мають бути в одному рядку
function getConfig() {
  return {    // ← фігурна дужка ОБОВ'ЯЗКОВО на цьому рядку
    host: 'localhost'
  };
}

Область видимості функції

JavaScript
const appName = 'MyApp'; // зовнішня змінна

function displayInfo() {
  const version = '1.0';          // локальна — видима тільки тут
  console.log(appName, version);   // доступ до зовнішньої — OK
}

displayInfo();
// console.log(version); // ReferenceError — недоступна зовні

// Функція не змінює зовнішню змінну (якщо не мутує об'єкт)
let counter = 0;
function increment() {
  counter++; // ПОБІЧНИЙ ЕФЕКТ — модифікує зовнішній стан
}
increment();
console.log(counter); // 1

Чисті функції

Функція є чистою якщо: (1) для однакових аргументів завжди повертає однаковий результат, (2) не має побічних ефектів.

JavaScript
// Чиста функція — предбачувана, легко тестується
function calculateTotal(price, taxRate) {
  return price * (1 + taxRate);
}

// Нечиста: залежить від зовнішнього стану
let taxRate = 0.2;
function calculateTotalImpure(price) {
  return price * (1 + taxRate); // результат залежить від зовнішньої zminnoi
}

// Нечиста: побічний ефект — змінює зовнішній масив
function addItemImpure(items, item) {
  items.push(item); // мутує вхідний масив!
  return items;
}

// Чиста: повертає новий масив
function addItem(items, item) {
  return [...items, item];
}

Найкращі практики

  • Одна функція — одна відповідальність. Якщо назва функції містить «і», «або» — це ознака, що варто розбити на дві.
  • Описові імена. Ім'я функції має описувати що вона робить: getUserById, calculateDiscount, isValidEmail.
  • Префікси. get — повертає значення, set — встановлює, is/has/can — повертає boolean, create — створює.
  • Мало параметрів. До 3 параметрів — прийнятно. Більше — розгляньте options-об'єкт.
JavaScript
// АНТИПАТЕРН: багато параметрів
function createRequest(method, url, body, timeout, retries, headers) {}

// КРАЩЕ: options об'єкт
function createRequest(url, options = {}) {
  const {
    method  = 'GET',
    body    = null,
    timeout = 5000,
    retries = 3,
    headers = {},
  } = options;
  // ...
}

Підсумок

  • Function declaration піднімається — можна викликати до оголошення.
  • Відсутній аргумент → undefined. Дефолтні значення спрацьовують лише для undefined, не для null.
  • return зупиняє функцію. Без нього — undefined. ASI: відкриваюча дужка на тому ж рядку що й return.
  • Функції мають доступ до зовнішнього scope. Обмежуйте побічні ефекти.
  • Чиста функція: однаковий вхід → однаковий вихід, без побічних ефектів.
