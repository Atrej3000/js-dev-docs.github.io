Логічні оператори

Початковий Основи JavaScript ~ 11 хв

Логічні оператори ||, &&, ! і ?? є виразами, що повертають одне з операндів — не обов'язково true або false. Це ключова відмінність від логіки у статично типізованих мовах.

|| — логічне АБО

Повертає перший truthy-операнд або останній операнд якщо всі falsy. Виконання зупиняється одразу після першого truthy (short-circuit).

JavaScript
// Повертає перший truthy або останній
console.log(1 || 2);         // 1  — перший truthy
console.log(null || 2);      // 2  — null falsy, повертає 2
console.log(0 || '' || 3);   // 3  — всі falsy до 3
console.log(0 || null);      // null — обидва falsy, повертає останній
console.log(false || 0 || undefined || 'ok'); // "ok"

// Short-circuit: другий операнд не обчислюється якщо перший truthy
let x = 1;
true || (x = 2);    // x залишається 1 — права частина не виконується
console.log(x);     // 1

// Класичний патерн: значення за замовчуванням
function greet(name) {
  const displayName = name || 'Гість';
  return `Привіт, ${displayName}!`;
}
console.log(greet(''));       // "Привіт, Гість!" — порожній рядок falsy
console.log(greet('Admin')); // "Привіт, Admin!"
|| і falsy-значення

|| замінює будь-яке falsy значення — включно з 0, "" і false. Якщо ці значення є валідними (наприклад, кількість = 0), використовуйте ?? замість ||.

&&— логічне І

Повертає перший falsy-операнд або останній операнд якщо всі truthy. Виконання зупиняється одразу після першого falsy.

JavaScript
// Повертає перший falsy або останній
console.log(1 && 2);          // 2    — обидва truthy, повертає останній
console.log(null && 2);       // null — перший falsy
console.log(1 && 0 && 3);     // 0    — 0 є falsy
console.log('a' && 'b' && 'c'); // "c" — всі truthy

// Short-circuit: другий операнд не виконується якщо перший falsy
let y = 1;
null && (y = 2);
console.log(y); // 1

// Умовне виконання (замість if)
const user = { isAdmin: true };
user.isAdmin && performAdminAction();
// Еквівалент: if (user.isAdmin) performAdminAction()

// Ланцюг перевірок
const street = user && user.address && user.address.street;
// Якщо user або address — falsy, street буде цим falsy значенням
// Сучасний аналог: optional chaining
const streetModern = user?.address?.street;

! — логічне НЕ

Інвертує булеве значення операнда. Завжди повертає true або false.

JavaScript
console.log(!true);      // false
console.log(!false);     // true
console.log(!0);         // true  (0 falsy → !0 = true)
console.log(!'');        // true
console.log(!null);      // true
console.log(!'hello');   // false (непорожній рядок truthy)
console.log(![]);        // false (порожній масив truthy!)

// Подвійне заперечення — конвертує в boolean
console.log(!!0);        // false
console.log(!!'hello');  // true
console.log(!!null);     // false
// Альтернатива: Boolean(value)

Пріоритет логічних операторів

! має вищий пріоритет, ніж &&, який вищий за ||.

JavaScript
// ! > && > ||
console.log(!false && true || false);
// Обчислення: (!false) && true || false
//           = true && true || false
//           = true || false
//           = true

// Дужки для явності
const isValid = (age >= 18) && (hasConsent || isGuardianPresent);
const canAccess = isAuthenticated && (isAdmin || hasPermission('read'));

Практичні патерни

JavaScript
// 1. Ланцюг перевірок з &&
function processPayment(user, amount) {
  return user && user.wallet && user.wallet.balance >= amount && executePayment(user, amount);
}

// 2. Значення за замовчуванням з ||
const timeout = config.timeout || 5000;
const host    = process.env.HOST || 'localhost';

// 3. Умовний рендеринг (React-стиль)
const element = isLoading && '
Loading...
';

// 4. Toggles через логічне НЕ
let isVisible = false;
isVisible = !isVisible; // true
isVisible = !isVisible; // false

// 5. Перевірка кількох умов
const isWeekend = day === 'Saturday' || day === 'Sunday';
const canDrive  = hasLicense && !isSuspended && age >= 16;

Edge cases

JavaScript
// && і || повертають операнд, не boolean
console.log(typeof (1 && 2));     // "number" — не "boolean"
console.log(typeof (null || 'x')); // "string"

// Небезпечне скорочення
const value = 0;
const result = value || 100; // 100 — але 0 може бути валідним!
// Правильно для null/undefined:
const result2 = value ?? 100; // 0 — ?? не замінює 0

// Вкладений && повертає falsy першого falsy
console.log(null && undefined); // null (перший falsy)
console.log(0 && null);         // 0
console.log(false && 0);        // false

Підсумок

  • || повертає перший truthy або останній операнд. Short-circuit зупиняє на першому truthy.
  • && повертає перший falsy або останній операнд. Short-circuit зупиняє на першому falsy.
  • ! завжди повертає true або false. !! — конвертація в boolean.
  • Пріоритет: ! > && > ||.
  • || замінює всі falsy значення (включно з 0 і ""). Для null/undefined використовуйте ??.
