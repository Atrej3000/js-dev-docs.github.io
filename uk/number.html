Числа

ПочатковийТипи даних~ 16 хв

Тип number представляє як цілі числа, так і числа з плаваючою точкою за стандартом IEEE 754 double-precision. Розуміння обмежень цього формату критично для роботи з фінансовими даними та великими числами.

Спеціальні числові значення

JavaScript
// Нескінченність
console.log(1 / 0);          // Infinity
console.log(-1 / 0);         // -Infinity
console.log(isFinite(42));   // true
console.log(isFinite(Infinity)); // false

// NaN — результат некоректних операцій
console.log(0 / 0);          // NaN
console.log(+'abc');          // NaN
console.log(NaN === NaN);    // false — унікальна властивість
console.log(Number.isNaN(NaN)); // true — правильна перевірка
console.log(isNaN('text'));     // true — конвертує перед перевіркою!
console.log(Number.isNaN('text')); // false — не конвертує, точніший

Методи числа

JavaScript
const n = 3.14159;

// Форматування
console.log(n.toFixed(2));       // "3.14" — рядок, не число
console.log(n.toPrecision(4));   // "3.142"
console.log((1234567).toLocaleString('uk-UA')); // "1 234 567"

// Перевірка
console.log(Number.isInteger(42));     // true
console.log(Number.isInteger(42.0));   // true
console.log(Number.isInteger(42.1));   // false
console.log(Number.isFinite(Infinity)); // false
console.log(Number.isSafeInteger(Number.MAX_SAFE_INTEGER)); // true

// Межі
console.log(Number.MAX_SAFE_INTEGER); // 9007199254740991 (2^53 - 1)
console.log(Number.MIN_SAFE_INTEGER); // -9007199254740991
console.log(Number.MAX_VALUE);        // ~1.8e+308
console.log(Number.EPSILON);          // ~2.22e-16

parseInt і parseFloat

JavaScript
// parseInt зупиняється на першому нецифровому символі
parseInt('42px', 10);    // 42
parseInt('3.14', 10);    // 3
parseInt('0xFF', 16);    // 255
parseInt('', 10);        // NaN — порожній рядок
parseInt('abc', 10);     // NaN

// Завжди вказуйте основу системи числення!
parseInt('010');         // 10 (сучасні рушії)
parseInt('010', 8);      // 8 — явна вісімкова

// parseFloat — для дробових
parseFloat('3.14abc');   // 3.14
parseFloat('abc');       // NaN

Об'єкт Math

JavaScript
Math.round(4.5);   // 5  — округлення до найближчого
Math.floor(4.9);   // 4  — вниз
Math.ceil(4.1);    // 5  — вгору
Math.trunc(4.9);   // 4  — відкидає дробову частину (ES6)
Math.trunc(-4.9);  // -4 (відмінність від floor для від'ємних)

Math.abs(-5);      // 5
Math.max(1, 2, 3); // 3
Math.min(1, 2, 3); // 1
Math.pow(2, 10);   // 1024 (краще: 2 ** 10)
Math.sqrt(16);     // 4
Math.cbrt(27);     // 3 — кубічний корінь

// Випадкове число [0, 1)
Math.random();

// Ціле число від min до max (включно)
function randomInt(min, max) {
  return Math.floor(Math.random() * (max - min + 1)) + min;
}
console.log(randomInt(1, 6)); // кидання кубика

Точність IEEE 754

JavaScript
// Класична проблема дробових чисел
console.log(0.1 + 0.2);          // 0.30000000000000004
console.log(0.1 + 0.2 === 0.3);  // false

// Рішення 1: Number.EPSILON
function isEqual(a, b) {
  return Math.abs(a - b) < Number.EPSILON;
}
console.log(isEqual(0.1 + 0.2, 0.3)); // true

// Рішення 2: toFixed для відображення (не обчислення!)
console.log((0.1 + 0.2).toFixed(1)); // "0.3"

// Рішення 3: BigInt або сторонні бібліотеки для фінансів
// Ніколи не зберігайте гроші як float — використовуйте цілі копійки
const priceInCents = 999; // 9.99 UAH
const tax = Math.round(priceInCents * 0.2); // 200 (2.00 UAH)

Підсумок

  • IEEE 754: 0.1 + 0.2 !== 0.3. Для порівняння дробів — Number.EPSILON. Для грошей — цілі копійки.
  • Number.isNaN точніший за глобальний isNaN (не конвертує аргумент).
  • parseInt(str, 10) — завжди з основою. Зупиняється на першому нецифровому.
  • Number.MAX_SAFE_INTEGER = 2^53 - 1. Для більших — BigInt.
  • Math.trunc vs Math.floor: для від'ємних дають різні результати.
