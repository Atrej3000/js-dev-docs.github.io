Робота з помилками: try...catch

СереднійОбробка помилок~ 14 хв

try...catch — механізм перехоплення помилок виконання. Синхронний код у блоці try виконується, і якщо виникає виняток — виконання переходить до catch. Блок finally виконується завжди.

Синтаксис

JavaScript
try {
  const data = JSON.parse(invalidJson);
  processData(data);
} catch (error) {
  console.error('Parsing failed:', error.message);
} finally {
  cleanup(); // виконується завжди — і при успіху, і при помилці
}

// catch без параметра (ES2019)
try {
  riskyOperation();
} catch {
  console.log('something went wrong');
}

Об'єкт Error

JavaScript
try {
  null.property; // TypeError
} catch (e) {
  console.log(e instanceof TypeError); // true
  console.log(e.name);                 // "TypeError"
  console.log(e.message);              // "Cannot read properties of null"
  console.log(e.stack);                // стек викликів
}

// Вбудовані типи помилок
new Error('base error');
new TypeError('wrong type');
new RangeError('out of range');
new ReferenceError('not defined');
new SyntaxError('invalid syntax');
new URIError('bad URI');

// throw будь-якого значення (але Error краще)
throw new Error('Something failed');
throw 42;       // дозволено, але не рекомендовано
throw 'oops';   // не рекомендовано — немає stack trace

Власні класи помилок

JavaScript
class ValidationError extends Error {
  constructor(field, message) {
    super(message);
    this.name  = 'ValidationError';
    this.field = field;
  }
}

class HttpError extends Error {
  constructor(status, message) {
    super(message);
    this.name   = 'HttpError';
    this.status = status;
  }
}

function processUser(data) {
  if (!data.name) throw new ValidationError('name', 'Name is required');
  if (!data.email) throw new ValidationError('email', 'Email is required');
  return saveUser(data);
}

try {
  processUser({ name: '' });
} catch (e) {
  if (e instanceof ValidationError) {
    console.log(`Validation failed on field "${e.field}": ${e.message}`);
  } else if (e instanceof HttpError) {
    console.log(`HTTP ${e.status}: ${e.message}`);
  } else {
    throw e; // перекидаємо невідомі помилки вище
  }
}

Перекидання помилок

JavaScript
// Обробляйте лише те, що знаєте — решту перекидайте
function loadConfig(path) {
  try {
    return JSON.parse(fs.readFileSync(path, 'utf-8'));
  } catch (e) {
    if (e instanceof SyntaxError) {
      throw new Error(`Invalid JSON in ${path}: ${e.message}`);
    }
    throw e; // ENOENT та інші — не наша відповідальність
  }
}

// finally для ресурсів (виконується незалежно від помилки)
function withConnection(fn) {
  const conn = openConnection();
  try {
    return fn(conn);
  } finally {
    conn.close(); // завжди закриваємо з'єднання
  }
}

Помилки в асинхронному коді

JavaScript
// try/catch не перехоплює помилки в setTimeout!
try {
  setTimeout(() => { throw new Error('async error'); }, 100);
} catch (e) {
  // НІКОЛИ не спрацює
}

// Правильно для async/await:
async function fetchData() {
  try {
    const res = await fetch('/api/data');
    if (!res.ok) throw new HttpError(res.status, res.statusText);
    return await res.json();
  } catch (e) {
    if (e instanceof HttpError) handleHttpError(e);
    else throw e;
  }
}

Підсумок

  • try/catch/finally: catch — при помилці, finally — завжди.
  • Об'єкт помилки: name, message, stack.
  • Власні класи помилок через extends Error — для типізованої обробки.
  • Перекидайте throw e якщо не знаєте як обробити помилку.
  • try/catch не перехоплює асинхронні помилки — для Promise використовуйте .catch() або async/await.
