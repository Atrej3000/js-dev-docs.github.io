Рекурсія та стек

СереднійФункції~ 15 хв

Рекурсія — техніка коли функція викликає саму себе. Кожен виклик додає новий кадр у стек викликів. Рекурсія є природною для задач зі вкладеною структурою (дерева, вкладені об'єкти) або математичних визначень через самоподібні підзадачі.

Базова структура

JavaScript
// Кожна рекурсивна функція має два випадки:
// 1. Base case — умова зупинки (без рекурсії)
// 2. Recursive case — виклик себе з меншим підзавданням

function factorial(n) {
  if (n <= 1) return 1;          // base case
  return n * factorial(n - 1);  // recursive case
}

console.log(factorial(5)); // 120
// factorial(5) = 5 * factorial(4)
//              = 5 * 4 * factorial(3)
//              = 5 * 4 * 3 * factorial(2)
//              = 5 * 4 * 3 * 2 * factorial(1)
//              = 5 * 4 * 3 * 2 * 1 = 120

Стек викликів

Кожен виклик функції резервує кадр у стеку: зберігає локальні змінні і адресу повернення. Рекурсія без base case або з надто великим входом спричинить Stack Overflow — вичерпання стеку.

JavaScript
// Stack overflow — нескінченна рекурсія
function infinite(n) {
  return infinite(n + 1); // немає base case
}
// RangeError: Maximum call stack size exceeded

// Типовий ліміт: ~10 000–15 000 кадрів залежно від рушія
// Перевірка глибини
function depth(n = 0) {
  try { return depth(n + 1); }
  catch (e) { return n; }
}
console.log(depth()); // ~12000–15000

Обхід вкладених структур

JavaScript
// Сума всіх чисел у вкладеному масиві довільної глибини
function deepSum(arr) {
  let total = 0;
  for (const item of arr) {
    if (Array.isArray(item)) {
      total += deepSum(item); // рекурсивно для підмасиву
    } else {
      total += item;
    }
  }
  return total;
}

console.log(deepSum([1, [2, [3, [4]], 5]])); // 15

// Обхід дерева (вкладений об'єкт)
function sumTree(node) {
  if (!node) return 0;
  return node.value + sumTree(node.left) + sumTree(node.right);
}

const tree = {
  value: 1,
  left:  { value: 2, left: { value: 4, left: null, right: null }, right: null },
  right: { value: 3, left: null, right: null },
};
console.log(sumTree(tree)); // 10

Мемоізація рекурсії

JavaScript
// Наївний Fibonacci: O(2^n) — дуже повільно
function fib(n) {
  if (n <= 1) return n;
  return fib(n - 1) + fib(n - 2);
}

// З мемоізацією: O(n)
function fibMemo(n, memo = {}) {
  if (n in memo) return memo[n];
  if (n <= 1) return n;
  memo[n] = fibMemo(n - 1, memo) + fibMemo(n - 2, memo);
  return memo[n];
}

console.log(fibMemo(50)); // 12586269025 — швидко

Підсумок

  • Рекурсія: base case (зупинка) + recursive case (менша задача).
  • Кожен виклик = новий кадр стеку. Ліміт ~10 000–15 000 в V8.
  • Природна для вкладених структур (дерева, вкладені масиви).
  • Мемоізація усуває повторні обчислення: O(2^n) → O(n) для Fibonacci.
  • Для лінійних задач (факторіал, сума) цикл ефективніший за рекурсію.
