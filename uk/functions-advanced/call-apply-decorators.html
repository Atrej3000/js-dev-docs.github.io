Декоратори та переадресація виклику, call/apply

СереднійФункції~ 14 хв

Декоратор функції — обгортка що розширює поведінку вихідної функції без зміни її коду. call і apply дозволяють викликати функцію з явним this — ключові інструменти для переадресації і запозичення методів.

call і apply

JavaScript
function greet(greeting, punctuation) {
  return `${greeting}, ${this.name}${punctuation}`;
}

const user = { name: 'Олена' };

// call: явний this + аргументи через кому
greet.call(user, 'Привіт', '!');    // "Привіт, Олена!"

// apply: явний this + аргументи масивом
greet.apply(user, ['Доброго дня', '.']); // "Доброго дня, Олена."

// Практичне: spread заміняє apply для Math.max
const nums = [3, 1, 4, 1, 5, 9];
Math.max.apply(null, nums); // 9 — старий стиль
Math.max(...nums);          // 9 — сучасний (рекомендований)

// Запозичення методу: Array-like → масив
function example() {
  const args = Array.prototype.slice.call(arguments);
  // або: const args = [...arguments];
  return args.map(x => x * 2);
}
console.log(example(1, 2, 3)); // [2, 4, 6]

Декоратори функцій

JavaScript
// Декоратор spy: записує всі виклики
function spy(fn) {
  function wrapper(...args) {
    wrapper.calls.push(args);
    return fn.apply(this, args); // зберігаємо this і аргументи
  }
  wrapper.calls = [];
  return wrapper;
}

function work(x) { return x * 2; }
const spiedWork = spy(work);

spiedWork(1);
spiedWork(2);
spiedWork(3);
console.log(spiedWork.calls); // [[1], [2], [3]]

// Декоратор delay: відкладає виконання
function delay(fn, ms) {
  return function(...args) {
    setTimeout(() => fn.apply(this, args), ms);
  };
}
const delayedLog = delay(console.log, 1000);
delayedLog('через секунду'); // виведе через 1000ms

// Декоратор debounce: виконує лише після паузи
function debounce(fn, ms) {
  let timer;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), ms);
  };
}
const debouncedSearch = debounce(search, 300);
// search викличеться лише якщо пауза між викликами > 300ms

Throttle декоратор

JavaScript
// throttle: не частіше ніж раз на ms мілісекунд
function throttle(fn, ms) {
  let lastCall = 0;
  return function(...args) {
    const now = Date.now();
    if (now - lastCall >= ms) {
      lastCall = now;
      return fn.apply(this, args);
    }
  };
}

// scroll handler: максимум раз на 200ms
window.addEventListener('scroll', throttle(() => {
  console.log('scroll position:', window.scrollY);
}, 200));

Підсумок

  • fn.call(ctx, ...args) — виклик з явним this, аргументи через кому.
  • fn.apply(ctx, [args]) — аргументи масивом. В сучасному коді замінюється spread.
  • Декоратор — обгортка що розширює функцію. fn.apply(this, args) — для «прозорої» передачі контексту і аргументів.
  • Spy — логує виклики. Delay — відкладає. Debounce — чекає паузу. Throttle — обмежує частоту.
