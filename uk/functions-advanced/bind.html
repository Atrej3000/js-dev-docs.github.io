Прив'язка контексту до функції

СереднійФункції~ 12 хв

Метод bind створює нову функцію з назавжди прив'язаним this і необов'язково частково застосованими аргументами. На відміну від call/apply, bind не викликає функцію — лише повертає прив'язану копію.

Прив'язка this

JavaScript
function greet(greeting) {
  return `${greeting}, ${this.name}!`;
}

const user = { name: 'Олена' };

// bind: повертає нову функцію з прив'язаним this
const boundGreet = greet.bind(user);
console.log(boundGreet('Привіт'));    // "Привіт, Олена!"
console.log(boundGreet('Доброго дня')); // "Доброго дня, Олена!"

// this більше не змінюється навіть через call/apply
console.log(boundGreet.call({ name: 'Інший' }, 'Hi')); // "Hi, Олена!"

// Класичний кейс: метод як callback
const timer = {
  id:    1,
  start() {
    setTimeout(this.tick.bind(this), 1000); // bind фіксує this = timer
  },
  tick() {
    console.log(`Timer ${this.id} tick`);
  }
};
timer.start(); // "Timer 1 tick" через 1 секунду

Часткове застосування (Partial Application)

JavaScript
function multiply(a, b) { return a * b; }

// bind(null, a) — фіксує перший аргумент, this не важливий
const double = multiply.bind(null, 2);
const triple = multiply.bind(null, 3);

console.log(double(5));  // 10
console.log(double(10)); // 20
console.log(triple(5));  // 15

// Практичне: лог-функції з префіксом
function log(level, message) {
  console.log(`[${level}] ${message}`);
}

const info  = log.bind(null, 'INFO');
const error = log.bind(null, 'ERROR');

info('Server started');   // [INFO] Server started
error('Connection lost'); // [ERROR] Connection lost

Стрілкові функції vs bind

JavaScript
class Timer {
  constructor() { this.ticks = 0; }

  // СПОСІБ 1: bind у конструкторі
  // constructor() {
  //   this.tick = this.tick.bind(this);
  // }

  // СПОСІБ 2: стрілкове поле класу (ES2022)
  tick = () => {
    this.ticks++; // this завжди = екземпляр
    console.log(this.ticks);
  };

  start() {
    setInterval(this.tick, 1000);
  }
}

const t = new Timer();
t.start(); // 1, 2, 3...

Підсумок

  • fn.bind(ctx) → нова функція з постійним this = ctx. Виклик відкладений.
  • Прив'язаний this не можна перевизначити через call/apply.
  • Часткове застосування: fn.bind(null, arg1) фіксує перший аргумент.
  • Стрілкові class fields — сучасна альтернатива bind у класах.
