Каррінг

СереднійРізне~ 11 хв

Каррінг — трансформація функції з кількома аргументами у ланцюг функцій з одним аргументом. f(a, b, c) → f(a)(b)(c). Дозволяє частковий виклик і композицію функцій.

Ручний каррінг

JavaScript
// Звичайна функція
function add(a, b) { return a + b; }

// Каррінг вручну
function curriedAdd(a) {
  return function(b) {
    return a + b;
  };
}

const add5 = curriedAdd(5);
console.log(add5(3));  // 8
console.log(add5(10)); // 15

// Практичний приклад
const multiply = a => b => a * b;
const double   = multiply(2);
const triple   = multiply(3);
console.log([1,2,3,4,5].map(double)); // [2,4,6,8,10]

Загальна функція curry

JavaScript
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args); // всі аргументи зібрані — викликаємо
    }
    return function(...moreArgs) {
      return curried.apply(this, args.concat(moreArgs));
    };
  };
}

const curriedAdd = curry((a, b, c) => a + b + c);

console.log(curriedAdd(1)(2)(3));  // 6
console.log(curriedAdd(1, 2)(3));  // 6
console.log(curriedAdd(1)(2, 3));  // 6
console.log(curriedAdd(1, 2, 3));  // 6

Композиція функцій

JavaScript
// pipe: застосовує функції зліва направо
const pipe = (...fns) => x => fns.reduce((v, f) => f(v), x);

const processUser = pipe(
  u => u.trim(),
  u => u.toLowerCase(),
  u => ({ username: u, createdAt: Date.now() }),
);

console.log(processUser('  Alice  '));
// { username: 'alice', createdAt: ... }

Підсумок

  • Каррінг: f(a, b) → f(a)(b). Кожен виклик повертає нову функцію.
  • Частковий виклик: фіксуємо перші аргументи, отримуємо спеціалізовану функцію.
  • Загальна curry(fn) підтримує гнучке застосування аргументів.
  • pipe і compose будуються поверх часткового застосування.
