Промісифікація

СереднійПроміси, async/await~ 8 хв

Промісифікація (promisification) — перетворення функції з callback-інтерфейсом у функцію, що повертає Promise. Це стандартна техніка для роботи з legacy Node.js API.

Ручна промісифікація

JavaScript
// Оригінальний callback-стиль
fs.readFile('data.txt', 'utf8', (err, data) => {
  if (err) reject(err);
  else resolve(data);
});

// Промісифікована версія
function readFilePromise(path, encoding = 'utf8') {
  return new Promise((resolve, reject) => {
    fs.readFile(path, encoding, (err, data) => {
      if (err) reject(err);
      else resolve(data);
    });
  });
}

// Використання
const data = await readFilePromise('data.txt');

Універсальна функція promisify

JavaScript
function promisify(fn) {
  return function(...args) {
    return new Promise((resolve, reject) => {
      fn(...args, (err, result) => {
        if (err) reject(err);
        else resolve(result);
      });
    });
  };
}

// Застосування до будь-якого error-first callback
const readFile  = promisify(fs.readFile);
const writeFile = promisify(fs.writeFile);
const stat      = promisify(fs.stat);

const content = await readFile('file.txt', 'utf8');
const info    = await stat('file.txt');
console.log(info.size);

util.promisify (Node.js)

JavaScript
const { promisify } = require('util');
// або
import { promisify } from 'util';

const readFile  = promisify(fs.readFile);
const writeFile = promisify(fs.writeFile);
const exec      = promisify(require('child_process').exec);

// Навіть зручніше: fs.promises (Node.js 10+)
import { readFile, writeFile } from 'fs/promises';
const data = await readFile('file.txt', 'utf8');

Кілька результатів у callback

JavaScript
// Деякі API повертають кілька значень у callback
// exec(cmd, (err, stdout, stderr) => ...)

function promisifyMulti(fn) {
  return function(...args) {
    return new Promise((resolve, reject) => {
      fn(...args, (err, ...results) => {
        if (err) reject(err);
        else resolve(results); // масив результатів
      });
    });
  };
}

const exec = promisifyMulti(require('child_process').exec);
const [stdout, stderr] = await exec('ls -la');

В Node.js функції з util.promisify.custom Symbol можна промісифікувати зі спеціальною логікою. Наприклад, dns.lookup використовує цей механізм.

Підсумок

  • Promisification — стандартна техніка адаптації callback API до Promise
  • Node.js: util.promisify або вбудований fs/promises
  • Власний promisify: fn(...args, (err, result) => resolve/reject)
  • Для кількох результатів — resolve з масивом
